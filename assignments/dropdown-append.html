Client-Side Web Technologies · Lab assignment 17

Dropdown to text box (max 10)

Question: Create a dropdown list containing numbers. When the user clicks the → button, append the selected number to a text box. Allow a maximum of 10 numbers.

View source code (HTML + JavaScript)
<div class="row" style="align-items:end">
  <div><label for="sel">Select a number</label><select id="sel"></select></div>
  <div><button id="add" style="margin-top:0">&rarr;</button><button id="clear" class="alt" style="margin-top:0">Clear</button></div>
</div>
<label for="box">Selected numbers</label><input id="box" readonly>
<p id="count"></p>
<div id="msg" class="msg"></div>

<script>
const sel = document.getElementById("sel");
for (let i = 1; i <= 20; i++) sel.innerHTML += "<option>" + i + "</option>";
const box = document.getElementById("box"), msg = document.getElementById("msg");
const items = [];
function update() {
  box.value = items.join(", ");
  document.getElementById("count").textContent = items.length + " of 10 numbers added";
}
document.getElementById("add").addEventListener("click", function () {
  if (items.length >= 10) {
    msg.className = "msg show err"; msg.textContent = "Maximum of 10 numbers allowed."; return;
  }
  msg.className = "msg";
  items.push(sel.value);
  update();
});
document.getElementById("clear").addEventListener("click", function () { items.length = 0; msg.className = "msg"; update(); });
update();
</script>
