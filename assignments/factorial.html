Client-Side Web Technologies · Lab assignment 14

Factorial of a number

Question: Write a JavaScript program to calculate the factorial of a given number.

View source code (HTML + JavaScript)
<label for="num">Enter a non-negative integer</label>
<input id="num" type="number" min="0" value="5">
<button id="b">Calculate factorial</button>
<div id="out" class="msg"></div>

<script>
function factorial(n) {
  let result = 1n;                 // BigInt so large factorials stay exact
  for (let i = 2n; i <= BigInt(n); i++) result *= i;
  return result;
}
document.getElementById("b").addEventListener("click", function () {
  const v = document.getElementById("num").value;
  const out = document.getElementById("out");
  const n = Number(v);
  if (v === "" || !Number.isInteger(n) || n < 0 || n > 500) {
    out.className = "msg show err"; out.textContent = "Enter a whole number between 0 and 500."; return;
  }
  out.className = "msg show ok";
  out.textContent = n + "! = " + factorial(n).toString();
  out.style.overflowWrap = "anywhere";
});
</script>
