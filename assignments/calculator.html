Basic calculator

Question: Create a calculator using JavaScript to perform basic arithmetic operations.

View source code (HTML + JavaScript)
<div class="calc">
  <input id="display" value="0" readonly aria-label="Display">
  <div class="keys" id="keys"></div>
</div>

<script>
const display = document.getElementById("display");
let current = "0", previous = null, operator = null, fresh = true;
const layout = ["C","+/-","%","/","7","8","9","*","4","5","6","-","1","2","3","+","0",".","DEL","="];
const keys = document.getElementById("keys");
layout.forEach(k => {
  const b = document.createElement("button");
  b.textContent = k;
  b.className = "/*-+=".includes(k) ? "op" : (/[0-9.]/.test(k) ? "num" : "alt");
  b.addEventListener("click", () => press(k));
  keys.appendChild(b);
});
function compute(a, b, op) {
  if (op === "+") return a + b;
  if (op === "-") return a - b;
  if (op === "*") return a * b;
  if (op === "/") return b === 0 ? "Error" : a / b;
  return b;
}
function press(k) {
  if (current === "Error") { current = "0"; fresh = true; }
  if (/[0-9]/.test(k)) { current = (fresh || current === "0") ? k : current + k; fresh = false; }
  else if (k === ".") { if (fresh) { current = "0."; fresh = false; } else if (!current.includes(".")) current += "."; }
  else if (k === "C") { current = "0"; previous = null; operator = null; fresh = true; }
  else if (k === "DEL") { current = current.length > 1 ? current.slice(0, -1) : "0"; }
  else if (k === "+/-") { current = String(-Number(current)); }
  else if (k === "%") { current = String(Number(current) / 100); }
  else if (k === "=") {
    if (operator !== null) { current = String(compute(previous, Number(current), operator)); previous = null; operator = null; fresh = true; }
  } else {
    if (operator !== null && !fresh) { current = String(compute(previous, Number(current), operator)); }
    previous = Number(current); operator = k; fresh = true;
  }
  if (current !== "Error" && current.length > 12) current = String(Number(current).toPrecision(10) / 1);
  display.value = current;
}
</script>
