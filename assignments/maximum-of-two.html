Client-Side Web Technologies · Lab assignment 18

Maximum of two numbers

Question: Create a web page with two text boxes to accept numbers and display the maximum of the two numbers.

View source code (HTML + JavaScript)
<div class="row">
  <div><label for="a">First number</label><input id="a" type="number"></div>
  <div><label for="b">Second number</label><input id="b" type="number"></div>
</div>
<button id="go">Find maximum</button>
<div id="out" class="msg"></div>

<script>
function maximum(x, y) { return x > y ? x : y; }
document.getElementById("go").addEventListener("click", function () {
  const a = document.getElementById("a").value, b = document.getElementById("b").value;
  const out = document.getElementById("out");
  if (a === "" || b === "" || isNaN(a) || isNaN(b)) {
    out.className = "msg show err"; out.textContent = "Please enter two valid numbers."; return;
  }
  const x = Number(a), y = Number(b);
  out.className = "msg show ok";
  out.textContent = x === y ? "Both numbers are equal: " + x : "Maximum of " + x + " and " + y + " is " + maximum(x, y);
});
</script>
