Mouse events

Question: Write a JavaScript program demonstrating mouseup, onclick, dblclick, and mouseover events.

Interact with this box
Hover, click, double-click or release the mouse button
Events will appear here...
View source code (HTML + JavaScript)
<div class="box" id="box" onmouseup="log('mouseup')" onclick="log('onclick')" ondblclick="log('dblclick')" onmouseover="log('mouseover')">
  <div><strong>Interact with this box</strong><br>Hover, click, double-click or release the mouse button</div>
</div>
<div class="log" id="log">Events will appear here...</div>
<button class="alt" onclick="document.getElementById('log').innerHTML=''">Clear log</button>

<script>
let n = 0;
function log(name) {
  n++;
  const box = document.getElementById("box");
  const colours = {mouseup:"#fff4d6", onclick:"#e2f5e9", dblclick:"#fde4e4", mouseover:"#e5ecff"};
  box.style.background = colours[name];
  const l = document.getElementById("log");
  if (n === 1) l.innerHTML = "";
  l.innerHTML = "<div>" + n + ". " + name + " event fired at " + new Date().toLocaleTimeString() + "</div>" + l.innerHTML;
}
</script>
